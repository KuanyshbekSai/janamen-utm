Инструкция по сохранению UTM-меток в Tilda

Скрипт захвата UTM параметров для вставки в блок T123 (HTML-код) на сайтах janamen1.kz и janamenhealth.kz

Как работает скрипт в Tilda:
  1. Захватывает все UTM-метки из адресной строки (`utm_source`, `utm_medium`, `utm_campaign`, `utm_content`, `utm_term`, `utm_id`).
  2. Сохраняет их в `localStorage` и `cookies` на 30 дней, чтобы метки не терялись при переходе между страницами сайта.
  3. Автоматически заполняет скрытые поля форм Tilda перед отправкой заявки в AmoCRM.
Код для блока T123 (HTML-код в Tilda): 
<script>
(function() {
  function getQueryParam(name) {
    var match = RegExp('[?&]' + name + '=([^&]*)').exec(window.location.search);
    return match ? decodeURIComponent(match[1].replace(/\+/g, ' ')) : null;
  }
  var utms = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_content', 'utm_term', 'utm_id'];
  utms.forEach(function(u) {
    var v = getQueryParam(u);
    if (v) {
      try { localStorage.setItem(u, v); } catch(e){}
      document.cookie = u + '=' + encodeURIComponent(v) + '; path=/; max-age=' + (30*86400);
    }
  });
  document.addEventListener('DOMContentLoaded', function() {
    utms.forEach(function(u) {
      var val = getQueryParam(u) || (function(){ try{return localStorage.getItem(u);}catch(e){return null;} })();
      if (!val) return;
      var inputs = document.querySelectorAll('input[name="' + u + '"]');
      inputs.forEach(function(inp) { inp.value = val; });
    });
  });
})();
</script>
